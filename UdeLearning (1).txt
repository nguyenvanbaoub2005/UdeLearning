-- Core helper: auto touch updated_at
-- =========================================================
CREATE OR REPLACE FUNCTION fn_set_updated_at()
RETURNS TRIGGER AS $$
BEGIN
    NEW.updated_at := NOW();
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE OR REPLACE FUNCTION fn_create_wallet_for_user()
RETURNS TRIGGER AS $$
BEGIN
    INSERT INTO wallets(user_id, currency, balance)
    VALUES (NEW.id, 'VND', 0)
    ON CONFLICT (user_id) DO NOTHING;

    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE OR REPLACE FUNCTION fn_apply_wallet_topup()
RETURNS TRIGGER AS $$
DECLARE
    v_wallet_id BIGINT;
    v_balance_before NUMERIC(14,2);
    v_balance_after NUMERIC(14,2);
BEGIN
    IF NEW.status <> 'paid' THEN
        RETURN NEW;
    END IF;

    IF TG_OP = 'UPDATE' AND OLD.status = 'paid' THEN
        RETURN NEW;
    END IF;

    SELECT id, balance
    INTO v_wallet_id, v_balance_before
    FROM wallets
    WHERE user_id = NEW.user_id
    FOR UPDATE;

    IF v_wallet_id IS NULL THEN
        RAISE EXCEPTION 'Wallet not found for user %', NEW.user_id;
    END IF;

    UPDATE wallets
    SET balance = balance + NEW.amount,
        updated_at = NOW()
    WHERE id = v_wallet_id;

    v_balance_after := v_balance_before + NEW.amount;

    INSERT INTO wallet_transactions(
        wallet_id, tx_type, direction, amount, balance_before, balance_after,
        status, ref_type, ref_id, description, created_at
    )
    VALUES (
        v_wallet_id, 'topup', 'credit', NEW.amount, v_balance_before, v_balance_after,
        'success', 'wallet_topup', NEW.id, 'Nạp tiền vào ví', NOW()
    )
    ON CONFLICT DO NOTHING;

    IF NEW.paid_at IS NULL THEN
        UPDATE wallet_topups
        SET paid_at = NOW(),
            updated_at = NOW()
        WHERE id = NEW.id;
    END IF;

    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

DROP TRIGGER IF EXISTS trg_apply_wallet_topup ON wallet_topups;
CREATE TRIGGER trg_apply_wallet_topup
AFTER INSERT OR UPDATE OF status ON wallet_topups
FOR EACH ROW
EXECUTE FUNCTION fn_apply_wallet_topup();

CREATE OR REPLACE FUNCTION fn_normalize_wallet_payment()
RETURNS TRIGGER AS $$
BEGIN
    IF NEW.method <> 'wallet' THEN
        RAISE EXCEPTION 'Only wallet payment method is allowed';
    END IF;

    NEW.provider := COALESCE(NULLIF(TRIM(NEW.provider), ''), 'wallet_core');

    IF NEW.provider_txn_id IS NULL OR LENGTH(TRIM(NEW.provider_txn_id)) = 0 THEN
        NEW.provider_txn_id := CONCAT('wallet_', gen_random_uuid()::TEXT);
    END IF;

    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

DROP TRIGGER IF EXISTS trg_normalize_wallet_payment ON payments;
CREATE TRIGGER trg_normalize_wallet_payment
BEFORE INSERT OR UPDATE OF method, provider, provider_txn_id ON payments
FOR EACH ROW
EXECUTE FUNCTION fn_normalize_wallet_payment();

CREATE OR REPLACE FUNCTION fn_process_wallet_payment()
RETURNS TRIGGER AS $$
DECLARE
    v_user_id BIGINT;
    v_order_total NUMERIC(12,2);
    v_wallet_id BIGINT;
    v_wallet_currency CHAR(3);
    v_wallet_balance NUMERIC(14,2);
    v_order_currency CHAR(3);
BEGIN
    IF NEW.method <> 'wallet' OR NEW.status <> 'paid' THEN
        RETURN NEW;
    END IF;

    IF TG_OP = 'UPDATE' AND OLD.status = 'paid' THEN
        RETURN NEW;
    END IF;

    SELECT o.user_id, o.total_amount, o.currency
    INTO v_user_id, v_order_total, v_order_currency
    FROM orders o
    WHERE o.id = NEW.order_id;

    IF v_user_id IS NULL THEN
        RAISE EXCEPTION 'Order % not found for wallet payment', NEW.order_id;
    END IF;

    IF NEW.amount <> v_order_total THEN
        RAISE EXCEPTION 'Wallet payment amount % must equal order total %', NEW.amount, v_order_total;
    END IF;

    SELECT w.id, w.currency, w.balance
    INTO v_wallet_id, v_wallet_currency, v_wallet_balance
    FROM wallets w
    WHERE w.user_id = v_user_id
      AND w.is_active = TRUE
    FOR UPDATE;

    IF v_wallet_id IS NULL THEN
        RAISE EXCEPTION 'Active wallet not found for user %', v_user_id;
    END IF;

    IF v_wallet_currency <> v_order_currency THEN
        RAISE EXCEPTION 'Currency mismatch: wallet % vs order %', v_wallet_currency, v_order_currency;
    END IF;

    IF v_wallet_balance < NEW.amount THEN
        RAISE EXCEPTION 'Insufficient wallet balance: need %, have %', NEW.amount, v_wallet_balance;
    END IF;

    UPDATE wallets
    SET balance = balance - NEW.amount,
        updated_at = NOW()
    WHERE id = v_wallet_id;

    INSERT INTO wallet_transactions(
        wallet_id, tx_type, direction, amount, balance_before, balance_after,
        status, ref_type, ref_id, description, created_at
    )
    VALUES (
        v_wallet_id,
        'purchase',
        'debit',
        NEW.amount,
        v_wallet_balance,
        v_wallet_balance - NEW.amount,
        'success',
        'payment',
        NEW.id,
        'Thanh toán đơn hàng bằng ví',
        NOW()
    )
    ON CONFLICT DO NOTHING;

    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

DROP TRIGGER IF EXISTS trg_process_wallet_payment ON payments;
CREATE TRIGGER trg_process_wallet_payment
BEFORE INSERT OR UPDATE OF status ON payments
FOR EACH ROW
EXECUTE FUNCTION fn_process_wallet_payment();

-- [A] Ensure course is publishable
CREATE OR REPLACE FUNCTION fn_validate_course_publish()
RETURNS TRIGGER AS $$
DECLARE
    v_lessons_count INT;
BEGIN
    IF NEW.status = 'published' AND COALESCE(OLD.status, 'draft') <> 'published' THEN
        SELECT COUNT(*) INTO v_lessons_count
        FROM lessons l
        JOIN course_sections cs ON cs.id = l.section_id
        WHERE cs.course_id = NEW.id
          AND l.deleted_at IS NULL;

        IF v_lessons_count = 0 THEN
            RAISE EXCEPTION 'Cannot publish course % without lessons', NEW.id;
        END IF;

        IF NEW.published_at IS NULL THEN
            NEW.published_at := NOW();
        END IF;
    END IF;

    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

DROP TRIGGER IF EXISTS trg_validate_course_publish ON courses;
CREATE TRIGGER trg_validate_course_publish
BEFORE UPDATE OF status ON courses
FOR EACH ROW
EXECUTE FUNCTION fn_validate_course_publish();

-- [B] Prevent duplicate purchase + optional auto-fill item price
CREATE OR REPLACE FUNCTION fn_prevent_duplicate_purchase()
RETURNS TRIGGER AS $$
DECLARE
    v_user_id BIGINT;
    v_course_price NUMERIC(12,2);
BEGIN
    SELECT o.user_id INTO v_user_id
    FROM orders o
    WHERE o.id = NEW.order_id;

    IF v_user_id IS NULL THEN
        RAISE EXCEPTION 'Order % does not exist.', NEW.order_id;
    END IF;

    IF EXISTS (
        SELECT 1
        FROM enrollments e
        WHERE e.user_id = v_user_id
          AND e.course_id = NEW.course_id
    ) THEN
        RAISE EXCEPTION 'User % already enrolled in course %', v_user_id, NEW.course_id;
    END IF;

    IF EXISTS (
        SELECT 1
        FROM order_items oi
        JOIN orders o ON o.id = oi.order_id
        WHERE o.user_id = v_user_id
          AND oi.course_id = NEW.course_id
          AND o.status IN ('pending', 'paid')
    ) THEN
        RAISE EXCEPTION 'User % already has course % in another pending/paid order', v_user_id, NEW.course_id;
    END IF;

    IF NEW.price IS NULL OR NEW.price = 0 THEN
        SELECT c.price INTO v_course_price
        FROM courses c
        WHERE c.id = NEW.course_id
          AND c.deleted_at IS NULL;

        IF v_course_price IS NULL THEN
            RAISE EXCEPTION 'Course % not found or deleted', NEW.course_id;
        END IF;

        NEW.price := v_course_price;
    END IF;

    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

DROP TRIGGER IF EXISTS trg_prevent_duplicate_purchase ON order_items;
CREATE TRIGGER trg_prevent_duplicate_purchase
BEFORE INSERT ON order_items
FOR EACH ROW
EXECUTE FUNCTION fn_prevent_duplicate_purchase();

-- [C] Keep order totals consistent
CREATE OR REPLACE FUNCTION fn_recalculate_order_totals()
RETURNS TRIGGER AS $$
DECLARE
    v_order_id BIGINT;
    v_subtotal NUMERIC(12,2);
    v_discount_percent NUMERIC(5,2);
    v_discount NUMERIC(12,2);
BEGIN
    v_order_id := COALESCE(NEW.order_id, OLD.order_id);

    SELECT COALESCE(SUM(price), 0)
    INTO v_subtotal
    FROM order_items
    WHERE order_id = v_order_id;

    SELECT c.discount_percent
    INTO v_discount_percent
    FROM orders o
    LEFT JOIN coupons c ON c.id = o.coupon_id
    WHERE o.id = v_order_id;

    v_discount := ROUND(v_subtotal * COALESCE(v_discount_percent, 0) / 100.0, 2);

    UPDATE orders
    SET subtotal_amount = v_subtotal,
        discount_amount = v_discount,
        total_amount = v_subtotal - v_discount,
        updated_at = NOW()
    WHERE id = v_order_id;

    RETURN COALESCE(NEW, OLD);
END;
$$ LANGUAGE plpgsql;

DROP TRIGGER IF EXISTS trg_recalculate_order_totals ON order_items;
CREATE TRIGGER trg_recalculate_order_totals
AFTER INSERT OR UPDATE OR DELETE ON order_items
FOR EACH ROW
EXECUTE FUNCTION fn_recalculate_order_totals();

-- [D] Protect payment state transition
CREATE OR REPLACE FUNCTION fn_validate_payment_transition()
RETURNS TRIGGER AS $$
BEGIN
    IF TG_OP = 'UPDATE' THEN
        IF OLD.status = 'failed' AND NEW.status IN ('paid', 'partial_refunded', 'refunded') THEN
            RAISE EXCEPTION 'Payment state transition from failed is not allowed';
        END IF;

        IF OLD.status = 'refunded' AND NEW.status <> 'refunded' THEN
            RAISE EXCEPTION 'Payment state transition from refunded is not allowed';
        END IF;

        IF OLD.status = 'partial_refunded' AND NEW.status = 'pending' THEN
            RAISE EXCEPTION 'Payment state transition partial_refunded -> pending is not allowed';
        END IF;

        IF OLD.status = 'partial_refunded' AND NEW.status = 'paid' THEN
            RAISE EXCEPTION 'Payment state transition partial_refunded -> paid is not allowed';
        END IF;

        IF OLD.status = 'paid' AND NEW.status = 'pending' THEN
            RAISE EXCEPTION 'Payment state transition paid -> pending is not allowed';
        END IF;
    END IF;

    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE OR REPLACE FUNCTION fn_validate_refund_transition()
RETURNS TRIGGER AS $$
BEGIN
    IF TG_OP = 'UPDATE' THEN
        IF OLD.status = 'processed' AND NEW.status <> 'processed' THEN
            RAISE EXCEPTION 'Refund state transition from processed is not allowed';
        END IF;

        IF OLD.status = 'rejected' AND NEW.status <> 'rejected' THEN
            RAISE EXCEPTION 'Refund state transition from rejected is not allowed';
        END IF;

        IF OLD.status = 'pending' AND NEW.status NOT IN ('approved', 'rejected', 'pending') THEN
            RAISE EXCEPTION 'Refund transition pending -> % is not allowed', NEW.status;
        END IF;

        IF OLD.status = 'approved' AND NEW.status NOT IN ('approved', 'processed') THEN
            RAISE EXCEPTION 'Refund transition approved -> % is not allowed', NEW.status;
        END IF;
    END IF;

    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE OR REPLACE FUNCTION fn_process_admin_refund()
RETURNS TRIGGER AS $$
DECLARE
    v_payment RECORD;
    v_order RECORD;
    v_wallet_id BIGINT;
    v_balance_before NUMERIC(14,2);
    v_balance_after NUMERIC(14,2);
    v_refunded_before NUMERIC(12,2);
    v_refunded_after NUMERIC(12,2);
    v_remaining_amount NUMERIC(12,2);
    v_payment_status TEXT;
    v_coupon_id BIGINT;
BEGIN
    IF NEW.status <> 'approved' THEN
        RETURN NEW;
    END IF;

    IF TG_OP = 'UPDATE' AND OLD.status = 'approved' THEN
        RETURN NEW;
    END IF;

    IF NEW.approved_by IS NULL THEN
        RAISE EXCEPTION 'Admin refund requires approved_by';
    END IF;

    IF NOT EXISTS (
        SELECT 1
        FROM user_roles ur
        JOIN roles r ON r.id = ur.role_id
        WHERE ur.user_id = NEW.approved_by
          AND r.name = 'admin'
    ) THEN
        RAISE EXCEPTION 'approved_by user % is not admin', NEW.approved_by;
    END IF;

    IF NOT EXISTS (
        SELECT 1
        FROM users u
        WHERE u.id = NEW.requested_by
    ) THEN
        RAISE EXCEPTION 'requested_by user % does not exist', NEW.requested_by;
    END IF;

    SELECT p.* INTO v_payment
    FROM payments p
    WHERE p.id = NEW.payment_id
    FOR UPDATE;

    IF v_payment.id IS NULL THEN
        RAISE EXCEPTION 'Payment % not found for refund', NEW.payment_id;
    END IF;

    IF v_payment.status NOT IN ('paid', 'partial_refunded') THEN
        RAISE EXCEPTION 'Only paid or partial_refunded payment can be refunded. Current status: %', v_payment.status;
    END IF;

    SELECT o.* INTO v_order
    FROM orders o
    WHERE o.id = v_payment.order_id
    FOR UPDATE;

    IF v_order.id IS NULL THEN
        RAISE EXCEPTION 'Order not found for payment %', v_payment.id;
    END IF;

    IF NEW.requested_by <> v_order.user_id AND NOT EXISTS (
        SELECT 1
        FROM user_roles ur
        JOIN roles r ON r.id = ur.role_id
        WHERE ur.user_id = NEW.requested_by
          AND r.name = 'admin'
    ) THEN
        RAISE EXCEPTION 'requested_by user % is not order owner/admin', NEW.requested_by;
    END IF;

    SELECT COALESCE(SUM(pr.amount), 0)
    INTO v_refunded_before
    FROM payment_refunds pr
    WHERE pr.payment_id = NEW.payment_id
      AND pr.status = 'processed';

    v_refunded_after := v_refunded_before + NEW.amount;

    IF v_refunded_after > v_payment.amount THEN
        RAISE EXCEPTION 'Refund amount overflow: processed %, new %, payment total %', v_refunded_before, NEW.amount, v_payment.amount;
    END IF;

    v_remaining_amount := v_payment.amount - v_refunded_after;

    IF v_remaining_amount = 0 THEN
        v_payment_status := 'refunded';
    ELSE
        v_payment_status := 'partial_refunded';
    END IF;

    SELECT w.id, w.balance INTO v_wallet_id, v_balance_before
    FROM wallets w
    WHERE w.user_id = v_order.user_id
      AND w.is_active = TRUE
    FOR UPDATE;

    IF v_wallet_id IS NULL THEN
        RAISE EXCEPTION 'Active wallet not found for refund user %', v_order.user_id;
    END IF;

    UPDATE wallets
    SET balance = balance + NEW.amount,
        updated_at = NOW()
    WHERE id = v_wallet_id;

    v_balance_after := v_balance_before + NEW.amount;

    INSERT INTO wallet_transactions(
        wallet_id, tx_type, direction, amount, balance_before, balance_after,
        status, ref_type, ref_id, description, created_at
    )
    VALUES (
        v_wallet_id,
        'refund',
        'credit',
        NEW.amount,
        v_balance_before,
        v_balance_after,
        'success',
        'payment_refund',
        NEW.id,
        CONCAT('Admin refund: ', NEW.reason),
        NOW()
    );

    UPDATE payments
    SET status = v_payment_status,
        updated_at = NOW()
    WHERE id = v_payment.id;

    UPDATE orders
    SET status = CASE WHEN v_payment_status = 'refunded' THEN 'refunded' ELSE 'partial_refunded' END,
        updated_at = NOW()
    WHERE id = v_order.id;

    v_coupon_id := v_order.coupon_id;

    IF NEW.refund_mode = 'revoke_access' THEN
        DELETE FROM enrollments e
        USING order_items oi
        WHERE oi.order_id = v_order.id
          AND e.user_id = v_order.user_id
          AND e.course_id = oi.course_id;

                DELETE FROM lesson_progress lp
                USING lessons l, course_sections cs, order_items oi
                WHERE lp.user_id = v_order.user_id
                    AND lp.lesson_id = l.id
                    AND l.section_id = cs.id
                    AND cs.course_id = oi.course_id
                    AND oi.order_id = v_order.id;

                DELETE FROM course_progress cp
                USING order_items oi
                WHERE cp.user_id = v_order.user_id
                    AND cp.course_id = oi.course_id
                    AND oi.order_id = v_order.id;

                DELETE FROM reviews r
                USING order_items oi
                WHERE r.user_id = v_order.user_id
                    AND r.course_id = oi.course_id
                    AND oi.order_id = v_order.id;

                DELETE FROM watch_history wh
                USING lessons l, course_sections cs, order_items oi
                WHERE wh.user_id = v_order.user_id
                    AND wh.lesson_id = l.id
                    AND l.section_id = cs.id
                    AND cs.course_id = oi.course_id
                    AND oi.order_id = v_order.id;
    END IF;

        DELETE FROM instructor_earnings
        WHERE payment_id = v_payment.id;

        IF v_remaining_amount > 0 THEN
                INSERT INTO instructor_earnings (instructor_id, payment_id, course_id, amount, system_amount, created_at)
                SELECT
                        c.instructor_id,
                        v_payment.id,
                        oi.course_id,
                        ROUND((v_remaining_amount * oi.price / NULLIF(v_payment.amount, 0)) * 0.70, 2),
                        ROUND((v_remaining_amount * oi.price / NULLIF(v_payment.amount, 0)) * 0.30, 2),
                        NOW()
                FROM order_items oi
                JOIN courses c ON c.id = oi.course_id
                WHERE oi.order_id = v_order.id;
        END IF;

    UPDATE payment_refunds
    SET status = 'processed',
        processed_at = NOW(),
        updated_at = NOW()
    WHERE id = NEW.id;

    IF v_payment_status = 'refunded' AND v_coupon_id IS NOT NULL THEN
        UPDATE coupons
        SET used_count = GREATEST(used_count - 1, 0),
            updated_at = NOW()
        WHERE id = v_coupon_id;
    END IF;

    INSERT INTO activity_logs(user_id, action, ref_type, ref_id, metadata, created_at)
    VALUES (
        v_order.user_id,
        'admin_refund',
        'payment_refund',
        NEW.id,
        jsonb_build_object(
            'payment_id', v_payment.id,
            'amount', NEW.amount,
            'refund_mode', NEW.refund_mode,
            'approved_by', NEW.approved_by,
            'requested_by', NEW.requested_by,
            'payment_status_after', v_payment_status
        ),
        NOW()
    );

    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

DROP TRIGGER IF EXISTS trg_validate_refund_transition ON payment_refunds;
CREATE TRIGGER trg_validate_refund_transition
BEFORE UPDATE OF status ON payment_refunds
FOR EACH ROW
EXECUTE FUNCTION fn_validate_refund_transition();

-- [E] Auto enroll + revenue share + buy log when payment is paid
CREATE OR REPLACE FUNCTION fn_handle_payment_paid()
RETURNS TRIGGER AS $$
DECLARE
    v_user_id BIGINT;
    v_coupon_id BIGINT;
BEGIN
    IF NEW.status <> 'paid' THEN
        RETURN NEW;
    END IF;

    IF TG_OP = 'UPDATE' AND OLD.status = 'paid' THEN
        RETURN NEW;
    END IF;

    SELECT o.user_id, o.coupon_id INTO v_user_id, v_coupon_id
    FROM orders o
    WHERE o.id = NEW.order_id
    FOR UPDATE;

    IF v_user_id IS NULL THEN
        RAISE EXCEPTION 'Order % not found for payment %', NEW.order_id, NEW.id;
    END IF;

    UPDATE orders
    SET status = 'paid',
        paid_at = COALESCE(NEW.paid_at, NOW()),
        updated_at = NOW()
    WHERE id = NEW.order_id;

    IF NEW.paid_at IS NULL THEN
        UPDATE payments
        SET paid_at = NOW(),
            updated_at = NOW()
        WHERE id = NEW.id;
    END IF;

    IF v_coupon_id IS NOT NULL THEN
        UPDATE coupons
        SET used_count = used_count + 1,
            updated_at = NOW()
        WHERE id = v_coupon_id
          AND is_active = TRUE
          AND expiry_date > NOW()
          AND (max_uses IS NULL OR used_count < max_uses);

        IF NOT FOUND THEN
            RAISE EXCEPTION 'Coupon % is invalid, expired or max uses reached', v_coupon_id;
        END IF;
    END IF;

    INSERT INTO enrollments (user_id, course_id, enrolled_at, source)
    SELECT v_user_id, oi.course_id, NOW(), 'payment'
    FROM order_items oi
    WHERE oi.order_id = NEW.order_id
    ON CONFLICT (user_id, course_id) DO NOTHING;

    INSERT INTO instructor_earnings (instructor_id, payment_id, course_id, amount, system_amount, created_at)
    SELECT
        c.instructor_id,
        NEW.id,
        oi.course_id,
        ROUND(oi.price * 0.70, 2),
        ROUND(oi.price * 0.30, 2),
        NOW()
    FROM order_items oi
    JOIN courses c ON c.id = oi.course_id
    WHERE oi.order_id = NEW.order_id
    ON CONFLICT (payment_id, course_id) DO NOTHING;

    INSERT INTO activity_logs(user_id, action, ref_type, ref_id, metadata, created_at)
    VALUES (
        v_user_id,
        'buy_course',
        'order',
        NEW.order_id,
        jsonb_build_object('payment_id', NEW.id, 'amount', NEW.amount),
        NOW()
    );

    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

DROP TRIGGER IF EXISTS trg_handle_payment_paid ON payments;
CREATE TRIGGER trg_handle_payment_paid
AFTER INSERT OR UPDATE OF status ON payments
FOR EACH ROW
EXECUTE FUNCTION fn_handle_payment_paid();

-- [F] Ensure lesson_progress only for enrolled user + set completed_at
CREATE OR REPLACE FUNCTION fn_validate_lesson_progress()
RETURNS TRIGGER AS $$
DECLARE
    v_course_id BIGINT;
BEGIN
    SELECT cs.course_id INTO v_course_id
    FROM lessons l
    JOIN course_sections cs ON cs.id = l.section_id
    WHERE l.id = NEW.lesson_id;

    IF v_course_id IS NULL THEN
        RAISE EXCEPTION 'Lesson % not found', NEW.lesson_id;
    END IF;

    IF NOT EXISTS (
        SELECT 1
        FROM enrollments e
        WHERE e.user_id = NEW.user_id
          AND e.course_id = v_course_id
    ) THEN
        RAISE EXCEPTION 'User % is not enrolled in course %', NEW.user_id, v_course_id;
    END IF;

    IF NEW.is_completed = TRUE AND (TG_OP = 'INSERT' OR OLD.is_completed = FALSE) THEN
        NEW.completed_at := COALESCE(NEW.completed_at, NOW());
    END IF;

    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_validate_lesson_progress
BEFORE INSERT OR UPDATE OF is_completed, lesson_id, user_id ON lesson_progress
FOR EACH ROW
EXECUTE FUNCTION fn_validate_lesson_progress();

-- [G] Auto update course progress on lesson progress change
CREATE OR REPLACE FUNCTION fn_update_course_progress()
RETURNS TRIGGER AS $$
DECLARE
    v_course_id BIGINT;
    v_total_lessons INT;
    v_completed_lessons INT;
    v_percent NUMERIC(5,2);
BEGIN
    SELECT cs.course_id INTO v_course_id
    FROM lessons l
    JOIN course_sections cs ON cs.id = l.section_id
    WHERE l.id = NEW.lesson_id;

    IF v_course_id IS NULL THEN
        RETURN NEW;
    END IF;

    SELECT COUNT(*) INTO v_total_lessons
    FROM lessons l
    JOIN course_sections cs ON cs.id = l.section_id
    WHERE cs.course_id = v_course_id
      AND l.deleted_at IS NULL;

    IF v_total_lessons = 0 THEN
        RETURN NEW;
    END IF;

    SELECT COUNT(*) INTO v_completed_lessons
    FROM lesson_progress lp
    JOIN lessons l ON l.id = lp.lesson_id
    JOIN course_sections cs ON cs.id = l.section_id
    WHERE lp.user_id = NEW.user_id
      AND lp.is_completed = TRUE
      AND cs.course_id = v_course_id
      AND l.deleted_at IS NULL;

    v_percent := ROUND((v_completed_lessons::NUMERIC * 100.0) / v_total_lessons, 2);

    INSERT INTO course_progress(user_id, course_id, progress_percent, created_at, updated_at)
    VALUES (NEW.user_id, v_course_id, v_percent, NOW(), NOW())
    ON CONFLICT (user_id, course_id)
    DO UPDATE SET
        progress_percent = EXCLUDED.progress_percent,
        updated_at = NOW();

    IF v_percent >= 100 THEN
        INSERT INTO activity_logs(user_id, action, ref_type, ref_id, metadata, created_at)
        VALUES (
            NEW.user_id,
            'complete_course',
            'course',
            v_course_id,
            jsonb_build_object('progress_percent', v_percent),
            NOW()
        );
    END IF;

    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

DROP TRIGGER IF EXISTS trg_update_course_progress ON lesson_progress;
CREATE TRIGGER trg_update_course_progress
AFTER INSERT OR UPDATE OF is_completed ON lesson_progress
FOR EACH ROW
EXECUTE FUNCTION fn_update_course_progress();

-- [H] Recalculate course and instructor average rating
CREATE OR REPLACE FUNCTION fn_refresh_ratings()
RETURNS TRIGGER AS $$
DECLARE
    v_course_id BIGINT;
    v_instructor_id BIGINT;
BEGIN
    v_course_id := COALESCE(NEW.course_id, OLD.course_id);

    UPDATE courses c
    SET rating_avg = COALESCE((
        SELECT ROUND(AVG(r.rating)::NUMERIC, 2)
        FROM reviews r
        WHERE r.course_id = v_course_id
    ), 0),
        updated_at = NOW()
    WHERE c.id = v_course_id;

    SELECT c.instructor_id INTO v_instructor_id
    FROM courses c
    WHERE c.id = v_course_id;

    IF v_instructor_id IS NOT NULL THEN
        UPDATE instructor_profiles ip
        SET rating_avg = COALESCE((
            SELECT ROUND(AVG(r.rating)::NUMERIC, 2)
            FROM reviews r
            JOIN courses c ON c.id = r.course_id
            WHERE c.instructor_id = v_instructor_id
        ), 0),
            updated_at = NOW()
        WHERE ip.user_id = v_instructor_id;
    END IF;

    RETURN COALESCE(NEW, OLD);
END;
$$ LANGUAGE plpgsql;

DROP TRIGGER IF EXISTS trg_refresh_ratings ON reviews;
CREATE TRIGGER trg_refresh_ratings
AFTER INSERT OR UPDATE OR DELETE ON reviews
FOR EACH ROW
EXECUTE FUNCTION fn_refresh_ratings();

-- [K] Auto create quiz attempt when answering starts
CREATE OR REPLACE FUNCTION fn_auto_create_quiz_attempt()
RETURNS TRIGGER AS $$
DECLARE
    v_attempt_id BIGINT;
BEGIN
    IF NEW.attempt_id IS NULL THEN
        INSERT INTO quiz_attempts(user_id, quiz_id, score, started_at, created_at)
        VALUES (NEW.user_id, NEW.quiz_id, 0, NOW(), NOW())
        RETURNING id INTO v_attempt_id;

        NEW.attempt_id := v_attempt_id;
    ELSE
        IF NOT EXISTS (
            SELECT 1
            FROM quiz_attempts qa
            WHERE qa.id = NEW.attempt_id
              AND qa.user_id = NEW.user_id
              AND qa.quiz_id = NEW.quiz_id
        ) THEN
            RAISE EXCEPTION 'Invalid attempt_id % for user % and quiz %', NEW.attempt_id, NEW.user_id, NEW.quiz_id;
        END IF;
    END IF;

    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_auto_create_quiz_attempt
BEFORE INSERT ON attempt_answers
FOR EACH ROW
EXECUTE FUNCTION fn_auto_create_quiz_attempt();
